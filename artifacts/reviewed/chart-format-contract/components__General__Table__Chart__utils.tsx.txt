59:export function getFiscalPeriodString(
60:  value: string | Date,
61:  getParsedPeriod: CreateDefaultProps["getParsedPeriod"],
62:  chartType = "",
63:): string {
64:  const hasTime = isDateTime(value);
65:
66:  if (value instanceof Date) {
67:    const valueStr = value.toISOString().replace("T", " ");
68:    return valueStr.slice(0, 19).replace(" 00:00:00", "");
69:  }
70:
71:  if (!(isDate(value) || hasTime)) return value;
72:
73:  const dateObj = dayjs(value);
74:
75:  if (hasTime && dateObj.format("HH:mm:ss").includes("00:00:00")) {
76:    return dateObj.format("YYYY-MM-DD");
77:  }
78:
79:  const { isQuarterly, isAnnual } = getParsedPeriod();
80:  if (isQuarterly) {
81:    const newLine = chartType?.includes("Column") ? "\n" : " ";
82:    return `Q${dateObj.quarter()}${newLine}${dateObj.year()}`;
83:  }
84:
85:  if (isAnnual) return `FY ${dateObj.year()}`;
86:
87:  return value;
88:}

90:function getLgToSm(values: Record<string, any>, decimals?: number) {
91:  return Object.keys(values || {})

93:    .sort((a, b) => values[b] - values[a])
94:    .map((key) => ({ label: key, value: formatNumber(values[key], decimals) }));

399:type ProcessSelectedNodeParams = {
400:  data: Record<string, any>;
401:  widgetDefs: { [field: string]: WidgetColumnDefT };
402:  dates: string[];
403:  dateField: string;
404:  columns: (string | Column<any>)[];

413:export function processSelectedNode(params: ProcessSelectedNodeParams) {
414:  const {
415:    data,
416:    widgetDefs,
417:    dates,
418:    dateField,
419:    columns,
420:    rawRowData,
421:    rawColumns,
422:    isReallyTransposed,
423:    noIndex,
424:    indexField = "Index",
425:    hideColumns,
426:  } = params;
427:
428:  if (
429:    isReallyTransposed &&
430:    ["fiscal_period", "period"].some((item) => item === data?.[indexField])
431:  )
432:    return;
433:
434:  for (const key of Object.keys(data)) {
435:    if (hideColumns.includes(key)) continue;
436:    if (!noIndex && columns?.length === 1) {
437:      if (key !== indexField && !columns.includes(key)) {
438:        columns.push(key);
439:      }
440:    }
441:    if (key === indexField && !rawColumns.includes(data[key])) {
442:      rawColumns.push(data[key]);
443:    }
444:  }
445:
446:  if (isReallyTransposed) {
447:    const fieldName = data?.[indexField];
448:    for (const key of Object.keys(data)) {
449:      if (key === indexField || !dates.includes(key)) continue;
450:      const existing = rawRowData.findIndex((item) => item[dateField] === key);
451:      if (existing !== -1) {
452:        rawRowData[existing][fieldName] = data[key];
453:      } else {
454:        const obj = { [dateField]: key };
455:        obj[fieldName] = data[key];
456:        rawRowData.push(obj);
457:      }
458:    }
459:    return;
460:  }
461:
462:  const newData = {};
463:  for (const key of Object.keys(data)) {
464:    const colDef = widgetDefs[key];
465:    newData[key] = data[key];
466:
467:    if (
468:      !(
469:        ["normalizedPercent", "percent"].includes(colDef?.formatterFn) ||
470:        colDef?.renderFn?.some((fn) => ["titleCase", "hoverCard"].includes(fn))
471:      )
472:    ) {
473:      continue;
474:    }
475:
476:    const params = { value: data[key] } as any;
477:    if (colDef?.renderFn?.includes("hoverCard") && typeof params.value === "object") {
478:      const { hoverCard } = colDef.renderFnParams;
479:      const cellField = hoverCard?.cellField || "value";
480:      params.value = params?.value?.[cellField] ?? "";
481:    }
482:
483:    const value = colDefFormatter(params, { colDef })?.value || params.value;
484:    newData[key] = value;
485:    if (typeof data[key] === "number") {
486:      newData[key] = Number(value.replace(/[^0-9.-]+/g, "").trim());
487:    }
488:  }
489:
490:  rawRowData.push(newData);
491:}

506:    selectedRowNodes,
507:    columnDefs,
508:    chartType,
509:    columns,

511:    isChartView = false,
512:  } = params;
513:
514:  const hasCustomChart =
515:    !widget?.external && customChartWidgetIds.includes(widget?.widgetId);
516:
517:  const widgetTable = widget?.data?.table || {};
518:  const widgetColDefs = widgetTable?.columnsDefs ?? [];
519:
520:  const storage = getWidgetStorage(widget);
521:  const period = storage?.period ?? storage?.params?.period;
522:  const transpose = hasCustomChart ? true : storage?.transpose;
523:
524:  const widgetDefs = Object.fromEntries(
525:    widgetColDefs?.map((column) => [
526:      column.field,
527:      {
528:        field: column.field,
529:        headerName: column.headerName,
530:        cellDataType: column.cellDataType,
531:        chartDataType: column.chartDataType,
532:        renderFn: column.renderFn,
533:        formatterFn: column.formatterFn,
534:        renderFnParams: column.renderFnParams || {},
535:        rowGroup: column.rowGroup,
536:        hide: column.hide,
537:        headerTooltip: column.headerTooltip,
538:        maxWidth: column.maxWidth,
539:        minWidth: column.minWidth,
540:        aggFunc: column.aggFunc,
541:        pinned: column.pinned,
542:      } as WidgetColumnDefT,
543:    ]),
544:  );
545:
546:  const hideColumns = widgetColDefs?.reduce((acc, curr) => {
547:    if (curr.hide || curr.chartDataType === "excluded") acc.push(curr.field);
548:    return acc;
549:  }, [] as string[]);
550:
551:  const dateField =
552:    widgetColDefs?.find(
553:      (column) => column.headerName === "Date" || column.cellDataType === "date",
554:    )?.field ?? "date";
555:
556:  const allKeys = Object.keys(selectedRowNodes[0] || {});
557:  const hasDates = allKeys.includes(dateField);
558:
559:  // In case the table is transposed and selectedNodes has only dates
560:  let indexField = "Index";
561:  let noIndex = !columns.includes("Index");
562:  const rowDataHasIndex = allKeys?.some((key) => key === "Index");
563:
564:  let isReallyTransposed = transpose && rowDataHasIndex;
565:  // If the widget is external, we determine whether it's really transposed
566:  // based on the presence of date fields and non-date fields, rather than relying on the storage value
567:  if (widget?.external) {
568:    const { nonDate, dateCols } = columns.reduce(
569:      (acc, key: string) => {
570:        if (isDate(key) || isDateTime(key)) acc.dateCols.push(key);
571:        else acc.nonDate.push(key);
572:        return acc;
573:      },
574:      { nonDate: [] as string[], dateCols: [] as string[] },
575:    );
576:
577:    if (nonDate.length === 1 && dateCols.length === columns.length - 1) {
578:      isReallyTransposed = true;
579:      indexField = nonDate[0];
580:      noIndex = false;
581:    }
582:  }
583:
584:  const dates = isReallyTransposed
585:    ? columns.filter((item) => item !== indexField)
586:    : hasDates && !isChartView
587:      ? selectedRowNodes.map((item) => item?.[dateField] || item?.date)
588:      : [];
589:
590:  const chartData = {
591:    columns,
592:    rowData: null,
593:    widgetId: widget?.widgetId,
594:    transpose: false,
595:    period,
596:    columnDefs,
597:    chartType: (dates?.length === 1 && ["area", "line"].includes(chartType)
598:      ? "groupedColumn"
599:      : chartType) as ChartType,
600:    isReallyTransposed,
601:  } as ChartDataT;
602:
603:  if (!(isReallyTransposed || hasCustomChart) && isChartView) return chartData;
604:
605:  const rawRowData = [] as Record<string, any>[];
606:  const rawColumns = (isReallyTransposed ? [] : columns) as string[];
607:
608:  const processData = (data: Record<string, any>) => {
609:    processSelectedNode({
610:      data,
611:      widgetDefs,
612:      dates,
613:      dateField,
614:      columns,
615:      rawRowData,
616:      rawColumns,
617:      isReallyTransposed,
618:      noIndex,
619:      indexField,
620:      hideColumns,
621:    });
622:  };
623:
624:  // Extract the data from the selected row nodes
625:  if (hasCustomChart || !isChartView)
626:    for (const data of selectedRowNodes) processData(data);
627:  else if (gridApi) {
628:    gridApi.forEachNodeAfterFilterAndSort((node) => {
629:      if (node.rowGroupColumn) {
630:        const autoColId = node.rowGroupColumn.getColId();
631:        if (!columns.includes(autoColId)) columns.splice(node.level, 0, autoColId);
632:      }
633:      getSelectedRangeData(node, undefined, processData);
634:    });
635:  }
636:  const needsColumnDefs =
637:    (widget?.external ? isReallyTransposed : transpose) ?? !columnDefs?.length;
638:
639:  if (
640:    !rawColumns.includes(dateField) &&
641:    Object.keys(rawRowData[0] || {}).includes(dateField)
642:  ) {
643:    rawColumns.unshift(dateField);
644:  }
645:
646:  if (rawColumns.includes(dateField)) {
647:    const dateSorted = gridApi?.getColumn(dateField)?.getSort();
648:    rawRowData?.sort((a, b) => {
649:      if (dateSorted === "desc") {
650:        return dayjs(b?.[dateField]).diff(dayjs(a?.[dateField]));
651:      }
652:      return dayjs(a?.[dateField]).diff(dayjs(b?.[dateField]));
653:    });
654:  }
655:
656:  return {
657:    ...chartData,
658:    columns: isReallyTransposed
659:      ? Array.from(
660:          new Set([
661:            dateField,
662:            ...Object.keys(rawRowData[0] || {}).filter(
663:              (col) => !hideColumns.includes(col),
664:            ),
665:          ]),
666:        )
667:      : rawColumns.filter((col) => !hideColumns.includes(col)),
668:    rowData: rawRowData,
669:    columnDefs: (needsColumnDefs
670:      ? getColumnDefs(rawRowData, {
671:          external: widget?.external,
672:          data: { table: { ...widgetTable, transpose: false } },
673:        } as WidgetT)
674:      : columnDefs
675:    )
676:      .filter((col) => rawRowData[0]?.[col.field] !== undefined)
677:      .sort((a, b) => {
678:        if (a.cellDataType === "date") return -1;
679:        if (b.cellDataType === "date") return 1;
680:        return 0;
681:      }),
682:  };
683:}
684:
685:type SeriesMapping = typeof CHART_TYPE_TO_SERIES_TYPE;
686:export type ChartSeriesT = SeriesMapping[ChartTypeExCombo];
687:
688:export function getSeriesType(chartType: ChartType): ChartSeriesT {
689:  return CHART_TYPE_TO_SERIES_TYPE[chartType as ChartTypeExCombo] ?? "line";
690:}
