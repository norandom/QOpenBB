364:export const colDefFormatter = (
365:  params: CellRendererProps,
366:  options: {
367:    colDef?: WidgetColumnDefT;
368:    widgetId?: string;
369:    decimalPlaces?: number;
370:  } = {},
371:): CellOnHoverProps => {
372:  const { colDef, widgetId, decimalPlaces: globalDecimalPlaces = 3 } = options;
373:  const decimalPlaces = colDef?.decimalPlaces ?? globalDecimalPlaces;
374:  const { colorValueKey, colorRules } = colDef?.renderFnParams ?? {};
375:
376:  const hasColorFns = colDef?.renderFn?.some((fn) =>
377:    ["columnColor", "greenRed"].includes(fn),
378:  );
379:  const color = hasColorFns || colorRules?.length > 0;
380:  const colorValue = color ? params?.data?.[colorValueKey] : undefined;
381:
382:  const colorProps = { color, colorValue, colorRules };
383:
384:  if (typeof params.value === "number" && colDef?.cellDataType !== "text") {
385:    colorProps.colorValue = colorValue ?? params.value;
386:
387:    let title = params.value.toLocaleString();
388:    let value = params.value.toString();
389:
390:    if (!colDef?.formatterFn || colDef?.formatterFn?.includes("int")) {
391:      value = formatNumber(params.value, decimalPlaces);
392:    }
393:    if (!colDef?.formatterFn || colDef?.formatterFn !== "none") {
394:      title = formatNumberThousands(params.value);
395:    }
396:    if (colDef?.formatterFn?.startsWith("normalized")) {
397:      const [int, decimals = "0"] = (params.value * 100).toString().split(".");
398:      const maxDecimals = params.value < 0.001 ? 6 : 2;
399:      value = `${int}.${decimals?.slice(0, Math.max(maxDecimals, decimalPlaces))}`;
400:
401:      if (value === "0.00") value = "0";
402:      if (colDef?.formatterFn === "normalizedPercent") {
403:        value = `${value} %`;
404:      }
405:    }
406:    if (colDef?.formatterFn === "percent") {
407:      const [int, decimals = "0"] = params.value.toString().split(".");
408:      value = `${int}.${decimals?.slice(0, Math.max(2, decimalPlaces))} %`;
409:    }
410:    if (
411:      !["normalizedPercent", "percent"].includes(colDef?.formatterFn) &&
412:      ["income_statement", "balance_sheet", "cash_flow_statement"].includes(widgetId)
413:    ) {
414:      if (params.value > 1_000 || params.value < -1_000) {
415:        value = formatNumberMagnitude(params.value, 0, true, true);
416:      }
417:    }
418:
419:    if (colDef?.prefix) {
420:      value = `${colDef.prefix}${value}`;
421:      title = `${colDef.prefix}${title}`;
422:    }
423:    if (colDef?.suffix) {
424:      value = `${value}${colDef.suffix}`;
425:      title = `${title}${colDef.suffix}`;
426:    }
427:
428:    return { value, title, ...colorProps };
429:  }
430:
431:  if (colDef?.renderFn?.includes("titleCase")) {
432:    const titleCase = convertToReadableLabel(params.value)
433:      .split(" ")
434:      .map((word: string) =>
435:        word
436:          .replace(/(\w{4,})/g, (match) => match.toLowerCase())
437:          .replace(/(\w{1})/, (match) => match.toUpperCase()),
438:      )
439:      .join(" ");
440:    return { value: titleCase, title: titleCase, ...colorProps };
441:  }
442:
443:  if (
444:    (colDef?.prefix || colDef?.suffix) &&
445:    params.value !== null &&
446:    params.value !== undefined
447:  ) {
448:    let stringValue = params.value.toString();
449:
450:    if (colDef?.prefix) {
451:      stringValue = `${colDef.prefix}${stringValue}`;
452:    }
453:    if (colDef?.suffix) {
454:      stringValue = `${stringValue}${colDef.suffix}`;
455:    }
456:
457:    return { value: stringValue, title: stringValue, ...colorProps };

459:
460:  if (colDef?.cellDataType === "date" && isDate(params.value)) {
461:    return { value: formatDate(params.value), ...colorProps };
462:  }

477:  const cellDataType = colDef?.cellDataType ?? params?.colDef?.cellDataType;
478:  const isDateField =
479:    (colField?.toString()?.includes("date") || cellDataType === "date") &&
480:    chartDataType !== "series";

485:    const value = params.value?.value ?? params.value;
486:    const defaultProps: CellOnHoverProps = { value };

491:      const dateObj = dayjs(value);
492:

494:        defaultProps.value = dateObj.format("YYYY-MM-DD");
495:      }
496:
497:      return defaultProps;
498:    }
499:
500:    if (isDateField && isDate(value)) {

505:        const year = calendar_year ? calendar_year : date.year();

515:      if (typeof params.value === "number") return defaultProps;
516:

518:    }

526:      ((isDateField || cellDataType === "text") &&
527:        !(colDef?.formatterFn || colDef?.renderFn))
528:    ) {
529:      return { value: formatObjectValue(value) };
530:    }

532:    return colDefFormatter({ ...params, value }, { widgetId, decimalPlaces, colDef });

709:    name: "Hide Column",
710:    action: () => {

713:  } as MenuItemDef;

716:    name: "Minimize Column",
