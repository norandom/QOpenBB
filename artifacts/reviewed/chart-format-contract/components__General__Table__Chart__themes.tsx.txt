128:export function createItemStyler(
129:  palette?: AgChartThemePalette,
130:  shouldFillBars = () => false,
131:) {
132:  const fills = palette?.fills || defaultFills;
133:
134:  // Track unique categories and their assigned colors
135:  const categoryColorMap = new Map<string, number>();
136:  let nextColorIndex = 0;
137:
138:  return (
139:    params: AgBarSeriesItemStylerParams<any, GridChartContext>,
140:  ): AgBarSeriesStyle => {
141:    if (!shouldFillBars?.()) return {};
142:
143:    // Only apply custom colors for single series (non-stacked) charts
144:    // Check if this is a stacked chart by looking at datum structure
145:    // Stacked charts have multiple yKey values across different series
146:    const datumKeys = Object.keys(params.datum || {});
147:    const hasMultipleSeriesKeys =
148:      datumKeys.filter(
149:        (key) =>
150:          key !== params.xKey &&
151:          key !== "date" &&
152:          key !== "node" &&
153:          typeof params.datum[key] === "number",
154:      ).length > 1;
155:
156:    // Let AG Charts handle stacked chart coloring
157:    if (hasMultipleSeriesKeys) return {};
158:
159:    // Determine if this is a bar or column chart and get the appropriate value
160:    const isBarChart = params.seriesId?.includes("BarSeries");
161:    const valueToUse = isBarChart ? params.xValue : params.yValue;
162:
163:    // Get category string for consistent mapping
164:    const categoryKey = valueToUse?.toString() || "";
165:
166:    // Check if we've seen this category before
167:    let colorIndex = categoryColorMap.get(categoryKey);
168:    if (colorIndex === undefined) {
169:      // Assign next sequential color
170:      colorIndex = nextColorIndex % fills.length;
171:      categoryColorMap.set(categoryKey, colorIndex);
172:      nextColorIndex++;
173:    }
174:
175:    return { fill: fills[colorIndex] };
176:  };
177:}

246:export function getCustomChartTheme(
247:  themeSettings: ThemeSettings["chartTheme"],
248:  shouldFillBars?: () => boolean,
249:  navigatorEnabled?: boolean,
250:  miniChartEnabled?: boolean,
251:): AgChartTheme<any, GridChartContext> {
252:  const isDarkMode = themeSettings.baseTheme === "ag-default-dark";
253:  const baseTheme = isDarkMode ? AgChartDarkTheme : AgChartLightTheme;
254:  const overrides = cloneDeep(baseTheme.overrides);
255:  const { backgroundImage, palette, params } = themeSettings;
256:
257:  const options = {
258:    ...baseTheme,
259:    palette,
260:    params,
261:    overrides,
262:  } as AgChartTheme<any, GridChartContext>;
263:
264:  let background: AgChartBackground | undefined;
265:  if (backgroundImage?.enabled && backgroundImage?.url) {
266:    const { enabled, ...imageConfig } = { ...backgroundImage };
267:
268:    background = { image: imageConfig };
269:  }
270:
271:  options.overrides.common.background = background;
272:  options.overrides.common.navigator = {
273:    enabled: Boolean(navigatorEnabled || miniChartEnabled),
274:    miniChart: { enabled: Boolean(miniChartEnabled) },
275:  };
276:
277:  // Add dynamic itemStyler with the theme's palette
278:  const itemStyler = createItemStyler(palette, shouldFillBars);
279:  options.overrides.bar.series.itemStyler = itemStyler;

281:  return options;

290:export function useAgThemes(theme: ThemeT) {
291:  const {
292:    chartBarFillEnabled = false,
293:    chartNavigatorEnabled = false,
294:    chartMiniChartEnabled = false,
295:  } = useWidgetContext(true)?.widget?.storage ?? {};
296:
297:  const shouldFillBars = useCallback(() => chartBarFillEnabled, [chartBarFillEnabled]);
298:
299:  const tableTheme = useShallowTableChartThemesStore(
300:    (state) => state[theme]?.tableTheme,
301:  );
302:  const agChartThemes = useShallowTableChartThemesStore((state) => ({
303:    "custom-dark": state.dark?.chartTheme,
304:    "custom-light": state.light?.chartTheme,
305:  }));
306:
307:  const agTheme = useMemo(() => {
308:    if (tableTheme?.params) {
309:      return themeAlpine
310:        .withPart(theme === "dark" ? colorSchemeDark : colorSchemeLight)
311:        .withPart(iconSetMap[tableTheme?.iconSet ?? "iconSetAlpine"])
312:        .withPart(customCssPart)
313:        .withParams({
314:          // deconstruct params because we might need to override some of them in the future
315:          ...tableTheme.params,
316:        });
317:    }
318:
319:    return getDefaultAgGridTheme(theme);
320:  }, [theme, tableTheme]);
321:
322:  const defaultChartThemesRef = useRef({
323:    "openbb-dark": getOpenBBChartTheme("dark"),
324:    "openbb-light": getOpenBBChartTheme("light"),
325:  });
326:
327:  const { themePalette, customChartThemes, chartThemes } = useMemo(() => {
328:    let themePalette: AgChartThemePalette | null = null;
329:    const updatedThemes = {} as Record<
330:      `custom-${ThemeT}` | `openbb-${ThemeT}`,
331:      AgChartTheme
332:    >;
333:    for (const [key, value] of Object.entries(agChartThemes ?? {})) {
334:      if (value?.baseTheme && value?.palette && value?.params) {
335:        themePalette = value.palette;
336:        updatedThemes[key] = getCustomChartTheme(
337:          value,
338:          shouldFillBars,
339:          chartNavigatorEnabled,
340:          chartMiniChartEnabled,
341:        );
342:        continue;
343:      }
344:      const obbKey = key.replace("custom-", "openbb-") as `openbb-${ThemeT}`;
345:      const obbTheme = defaultChartThemesRef.current[obbKey];
346:      obbTheme.overrides.bar.series.itemStyler = createItemStyler(null, shouldFillBars);
347:      obbTheme.overrides.common.navigator = {
348:        enabled: chartNavigatorEnabled || chartMiniChartEnabled,
349:        miniChart: { enabled: chartMiniChartEnabled },
350:      };
351:
352:      updatedThemes[obbKey] = obbTheme;
353:    }
354:
355:    let chartTheme = `custom-${theme}` as `custom-${ThemeT}` | `openbb-${ThemeT}`;
356:    if (!updatedThemes[chartTheme]) chartTheme = `openbb-${theme}`;
357:
358:    return {
359:      themePalette,
360:      customChartThemes: updatedThemes,
361:      chartThemes: [chartTheme],
362:    };
363:  }, [
364:    defaultChartThemesRef,
365:    agChartThemes,
366:    theme,
367:    shouldFillBars,
368:    chartNavigatorEnabled,
369:    chartMiniChartEnabled,
370:  ]);
371:
372:  return { themePalette, agTheme, chartThemes, customChartThemes };
373:}
