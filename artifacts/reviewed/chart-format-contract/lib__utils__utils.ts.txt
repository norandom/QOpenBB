49:  try {
50:    // Use linkify-it to find URLs in text
51:    const matches = linkify.match(text);
52:
53:    if (!matches) {
54:      return [];
55:    }
56:
57:    // Extract URLs and filter out emails (mailto: links) and @mentions
58:    const urls = matches
59:      .map((match) => match.url)
60:      .filter((url) => {
61:        if (url.endsWith(".name")) return false;
62:        // Filter out mailto: links (emails)
63:        if (url.startsWith("mailto:")) return false;
64:
65:        // Check if preceded by @ (indicating email domain pattern like @gmail.com)
66:        // https://openbb.atlassian.net/browse/AA-3812
67:        const originalText = matches.find((m) => m.url === url)?.raw || "";
68:        const urlIndex = text.indexOf(originalText);
69:        if (urlIndex > 0 && text[urlIndex - 1] === "@") {
70:          return false;
71:        }
72:
73:        return true;
74:      });

76:    return urls;

317:export function getContrastColor(hexColor) {
318:  // If the color is not a valid hex color, return black as default
319:  if (!/^#([0-9A-F]{3}){1,2}$/i.test(hexColor)) {
320:    return "#000000";
321:  }
322:
323:  // Convert the hex color to RGB
324:  let { r, g, b } = { r: 0, g: 0, b: 0 }; // linting hack
325:  if (hexColor.length === 7) {
326:    r = Number.parseInt(hexColor.substr(1, 2), 16);
327:    g = Number.parseInt(hexColor.substr(3, 2), 16);
328:    b = Number.parseInt(hexColor.substr(5, 2), 16);
329:  } else {
330:    r = Number.parseInt(hexColor[1] + hexColor[1], 16);
331:    g = Number.parseInt(hexColor[2] + hexColor[2], 16);
332:    b = Number.parseInt(hexColor[3] + hexColor[3], 16);
333:  }

335:  // Calculate the brightness of the color
336:  const brightness = (r * 299 + g * 587 + b * 114) / 1000;

339:  return brightness > 128 ? "#000000" : "#FFFFFF";

342:export function generateBackgroundColorWithContrast(
343:  textColor: "white" | "black" = "black",
344:  threshold = 4.5,
345:  vibrancy = 0.5,
346:): string {
347:  const getVibrancy = (color: string): number => {
348:    const getSaturation = (r: number, g: number, b: number): number => {
349:      const max = Math.max(r, g, b);
350:      const min = Math.min(r, g, b);
351:      const delta = max - min;
352:      const sum = max + min;
353:      return delta === 0 ? 0 : delta / sum;
354:    };
355:
356:    const rgb = hexToRGB(color);
357:    const saturation = getSaturation(...rgb);
358:    return saturation;
359:  };
360:
361:  const getContrastRatio = (color: string): number => {
362:    const rgbToLuminance = (r: number, g: number, b: number): number => {
363:      const sRGB = (c: number): number => {
364:        const channel = c / 255;
365:        return channel <= 0.03928
366:          ? channel / 12.92
367:          : // biome-ignore lint: Use the '**' operator instead of 'Math.pow'
368:            Math.pow((channel + 0.055) / 1.055, 2.4);
369:      };
370:
371:      const gammaRGB = [sRGB(r), sRGB(g), sRGB(b)];
372:      return 0.2126 * gammaRGB[0] + 0.7152 * gammaRGB[1] + 0.0722 * gammaRGB[2];
373:    };
374:
375:    const textColorLuminance = rgbToLuminance(...hexToRGB(textColor));
376:    const backgroundColorLuminance = rgbToLuminance(...hexToRGB(color));
377:    return (
378:      (Math.max(textColorLuminance, backgroundColorLuminance) + 0.05) /
379:      (Math.min(textColorLuminance, backgroundColorLuminance) + 0.05)
380:    );
381:  };
382:
383:  let backgroundColor = generateRandomColor();
384:  let contrastRatio = getContrastRatio(backgroundColor);
385:  let colorVibrancy = getVibrancy(backgroundColor);
386:
387:  while (contrastRatio < threshold || colorVibrancy < vibrancy) {
388:    backgroundColor = generateRandomColor();
389:    contrastRatio = getContrastRatio(backgroundColor);
390:    colorVibrancy = getVibrancy(backgroundColor);
391:  }
392:
393:  return backgroundColor;
394:}

441:export function slugifyWithExtension(str: string) {
442:  if (!str) return "";
443:
444:  // Convert to string if it isn't already
445:  const inputStr = str.toString();
446:
447:  // Find the last period to separate filename from extension
448:  const lastDotIndex = inputStr.lastIndexOf(".");
449:
450:  // If there's no dot or the dot is at the beginning or end, treat the whole string as filename
451:  if (lastDotIndex <= 0 || lastDotIndex === inputStr.length - 1) {
452:    // No extension, slugify the entire string
453:    return inputStr
454:      .toLowerCase()
455:      .trim()
456:      .replace(/\s+/g, "_") // Replace one or more spaces with single underscore
457:      .replace(/\n/g, "_")
458:      .replace(/[^a-zA-Z0-9_\u0080-\uFFFF.-]/g, "_") // Replace special characters with underscore, but keep dots for now
459:      .replace(/_+/g, "_") // Replace multiple consecutive underscores with single underscore
460:      .replace(/^_|_$/g, "") // Remove leading and trailing underscores
461:      .replace(/\.{2,}/g, "."); // Replace multiple consecutive dots with single dot
462:  }
463:
464:  // Split into filename and extension
465:  const filename = inputStr.substring(0, lastDotIndex);
466:  const extension = inputStr.substring(lastDotIndex + 1);
467:
468:  // Slugify the filename part
469:  const slugifiedFilename = filename
470:    .toLowerCase()
471:    .trim()
472:    .replace(/\s+/g, "_") // Replace one or more spaces with single underscore
473:    .replace(/\n/g, "_")
474:    .replace(/[^a-zA-Z0-9_\u0080-\uFFFF.-]/g, "_") // Replace special characters with underscore, but keep dots for now
475:    .replace(/_+/g, "_") // Replace multiple consecutive underscores with single underscore
476:    .replace(/^_|_$/g, "") // Remove leading and trailing underscores
477:    .replace(/\.{2,}/g, "."); // Replace multiple consecutive dots with single dot

480:  const slugifiedExtension = extension

482:    .trim()

502:export function beautifySlug(slug: string, toUpperCase = true): string {
503:  if (!slug) return "";
504:
505:  try {
506:    return slug
507:      .toString()
508:      .replace(/_/g, " ")
509:      .split(/(?=[A-Z][a-z])|(?<=[a-z])(?=[A-Z])|(?<=[A-Z])(?=[A-Z][a-z])/)
510:      .reduce((acc, word) => {
511:        // avoids adding space after opening parenthesis
512:        const addSpace = acc.endsWith("(") ? "" : " ";
513:        if (!toUpperCase) return `${acc}${addSpace}${word}`;
514:
515:        const firstChar = word.charAt(0).toUpperCase();
516:        acc += `${addSpace}${firstChar}${word.slice(1)}`;
517:        return acc;
518:      }, "");
519:  } catch (e) {
520:    return slug
521:      .toString()
522:      .replace(/_/g, " ")
523:      .split(/(?=[A-Z][a-z])|(?=[A-Z][a-z])/)
524:      .map((word) => {
525:        if (!toUpperCase) return word;
526:        return word.charAt(0).toUpperCase() + word.slice(1);
527:      })
528:      .join(" ");
529:  }

582:  if (!value?.toFixed) return value?.toString();
583:
584:  if (value % 1 !== 0) {
585:    if (value < 1 && value > -1) {
586:      decimals = Math.max(decimals, 2);

589:    const toFixed = Math.min(decimals, totalDecimals);

591:    const rounded = value?.toFixed(toFixed);
592:    return rounded?.padEnd(decimals + rounded?.indexOf(".") + 1, "0");
593:  }
594:  return value?.toString();

597:export function formatNumberMagnitude(
598:  value: number | string,
599:  decimals = 3,
600:  inMillions = false,
601:  noSuffix = false,
602:) {
603:  if (typeof value === "string") {
604:    value = Number(formatNumberNoMagnitude(value));
605:  }
606:
607:  if (value === 0) return "0";
608:  if (Number.isNaN(value)) return "N/A";
609:
610:  value = Number(toFixedString(value, decimals) || 0);
611:
612:  if (value > 1_000_000 || value < -1_000_000) {
613:    const magnitude = Math.min(
614:      inMillions ? 2 : 4,
615:      Math.floor(Math.log10(Math.abs(value)) / 3),
616:    );
617:    const suffix = [" ", "", "M", ...(inMillions ? [] : ["B", "T"])][magnitude];
618:    const formatted = round(
619:      value / 10 ** (magnitude * 3),
620:      Math.min(decimals, 3),
621:    ).toString();

1359:/**
1360: * Merges two objects recursively, removing any functions and merging objects.
1361: * @param target - The target object to merge into
1362: * @param source - The source object to merge from
1363: * @param withFunctions - Whether to include functions in the merged object (default: false)
1364: * @returns The merged object
1365: * @example
1366: * const target = { a: 1, b: { c: 2 } };
1367: * const source = { a: 2, b: { d: 3 } };
1368: * const merged = mergeObjects(target, source);
1369: * // merged = { a: 2, b: { c: 2, d: 3 } }
1370: *
1371: **/
1372:export function mergeObjects<T, U>(target: T, source: U, withFunctions = false): T & U {
1373:  try {
1374:    const sourceKeys = Object.keys(source || {});
1375:
1376:    for (const key of Object.keys(source || {})) {
1377:      let removeKey = false;
1378:      if (typeof source[key] === "object" && !Array.isArray(source[key])) {
1379:        if (!target?.[key]) Object.assign(target, { [key]: {} });
1380:
1381:        Object.assign(
1382:          target[key],
1383:          mergeObjects(target[key], source[key], withFunctions),
1384:        );
1385:        removeKey = true;
1386:      }
1387:      if (Array.isArray(source[key])) {
1388:        Object.assign(target, { [key]: source[key] });
1389:        removeKey = true;
1390:      }
1391:
1392:      removeKey && sourceKeys.splice(sourceKeys.indexOf(key), 1);
1393:    }
1394:
1395:    for (const key of sourceKeys) {
1396:      if (typeof source[key] !== "object") {
1397:        if (typeof source[key] !== "function" || withFunctions) {
1398:          Object.assign(target, { [key]: source[key] });
1399:        }
1400:      }
1401:    }
1402:
1403:    return target as T & U;
1404:  } catch (e) {
1405:    console.log(`Error in mergeObjects: ${e}`);
